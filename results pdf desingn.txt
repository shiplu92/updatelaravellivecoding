public function ResutlsController(Request $request){

$result=Studentmarked::where('year_id',$request->batch_id)
->where('class_id',$request->Semester_id)
->where('depart_id',$request->Department)
->where('examtype_id',$request->examtype_id)
->first();





if($result==true){


$allresult=Studentmarked::with(['examtype','dpi','spi','student','bpi'])->select('year_id','class_id','depart_id','examtype_id','student_id')
->where('year_id',$request->batch_id)
->where('class_id',$request->Semester_id)
->where('depart_id',$request->Department)
->where('examtype_id',$request->examtype_id)

->groupBy('year_id')->groupBy('class_id')->groupBy('depart_id')->groupBy('examtype_id')->groupBy('student_id')

->get();


$pdf=PDF::loadview('admin.Report.result.print',['allresult'=>$allresult]);
return $pdf->stream('document.pdf');

}else{
       $notification=array('messege' => ' sorry!', 'alert-type' => 'success');
        return redirect()->back()->with($notification);



}








}
