public function marksheetganarated(Request $request){


$singlestudent=Studentmarked::where('year_id',$request->batch_id)
->where('class_id',$request->Semester_id)
->where('depart_id',$request->Department)
->where('examtype_id',$request->examtype_id)
->where('student_id',$request->Student_id)


->first();




$count_fail =DB::table('marks')->where('year_id',$request->batch_id)
->where('class_id',$request->Semester_id)
->where('depart_id',$request->Department)
->where('examtype_id',$request->examtype_id)
->where('student_id',$request->Student_id)->where('marks','<','33')->get()->count();




if($singlestudent==true){
    $allstudent=Studentmarked::with(['examtype','dpi','spi','student','bpi'])

    ->where('year_id',$request->batch_id)
    ->where('class_id',$request->Semester_id)
->where('depart_id',$request->Department)
->where('examtype_id',$request->examtype_id)
->where('student_id',$request->Student_id)





->get();
$getpoin=Getpointed::all();




    return view('admin.marksheet.marksheetprint',compact('allstudent','getpoin','count_fail'));

}else{

        $notification=array('messege' => ' sorry!', 'alert-type' => 'success');
        return redirect()->back()->with($notification);



    //return redirect()->back()->with('error','sorry!thes critera does not match');
}






    
}
